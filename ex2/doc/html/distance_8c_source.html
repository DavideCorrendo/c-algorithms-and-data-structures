63static int edit_distance_memo(const char *s1, const char *s2, int i, int j, int **memo) {
64 if (i == 0) return j;
65 if (j == 0) return i;
66
67 if (memo[i][j] != -1) {
68 return memo[i][j];
69 }
70
71 // Calculate cost of operations
72 int d_noop = (s1[i - 1] == s2[j - 1]) ?
73 edit_distance_memo(s1, s2, i - 1, j - 1, memo) :
74 1 + edit_distance_memo(s1, s2, i - 1, j - 1, memo);
75
76 int d_canc = 1 + edit_distance_memo(s1, s2, i, j - 1, memo);
77 int d_ins = 1 + edit_distance_memo(s1, s2, i - 1, j, memo);
78
79 memo[i][j] = min3(d_noop, d_canc, d_ins);
80 return memo[i][j];
