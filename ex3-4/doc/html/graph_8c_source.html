138Bool graph_remove_node(Graph* gr, const void* node) {
139 if (!gr || !node) return false;
140
141 HashTable* adj = hash_table_get(gr->nodes, node);
142 if (!adj) return false;
143
144 // Remove all edges connected to this node
145 void** neighbors = hash_table_keyset(adj);
146 if (neighbors) {
147 for (int i = 0; neighbors[i] != NULL; i++) {
148 graph_remove_edge(gr, node, neighbors[i]);
149 if (!gr->is_directed) {
150 graph_remove_edge(gr, neighbors[i], node);
151 }
152 }
153 free(neighbors);
154 }
155
156 // Free adjacency list and remove node
157 hash_table_free(adj);
158 hash_table_remove(gr->nodes, node);
159
160 return true;

170Bool graph_remove_edge(Graph* gr, const void* node1, const void* node2) {
171 if (!gr || !node1 || !node2) return false;
172
173 HashTable* adj1 = hash_table_get(gr->nodes, node1);
174 if (!adj1) return false;
175
176 // Remove edge(s)
177 if (hash_table_contains_key(adj1, node2)) {
178 // Free label if graph is labelled
179 if (gr->is_labelled) {
180 void* label = hash_table_get(adj1, node2);
181 free(label);
182 }
183 hash_table_remove(adj1, node2);
184
185 // Remove reverse edge for undirected graphs
186 if (!gr->is_directed) {
187 HashTable* adj2 = hash_table_get(gr->nodes, node2);
188 if (adj2) {
189 if (gr->is_labelled) {
190 void* reverse_label = hash_table_get(adj2, node1);
191 free(reverse_label);
192 }
193 hash_table_remove(adj2, node1);
194 }
195 }
196
197 gr->edge_count--;
198 return true;
199 }
200
201 return false;

236Edge** graph_get_edges(const Graph* gr) {
237 if (!gr || gr->edge_count == 0) return NULL;
238
239 // Allocate array for edges
240 Edge** edges = malloc((gr->edge_count + 1) * sizeof(Edge*));
241 if (!edges) return NULL;
242
243 int index = 0;
244 void** nodes = hash_table_keyset(gr->nodes);
245 if (!nodes) {
246 free(edges);
247 return NULL;
248 }
249
250 // Iterate through all nodes and their edges
251 for (int i = 0; nodes[i] != NULL; i++) {
252 HashTable* adj = hash_table_get(gr->nodes, nodes[i]);
253 void** neighbors = hash_table_keyset(adj);
254
255 if (neighbors) {
256 for (int j = 0; neighbors[j] != NULL; j++) {
257 Edge* edge = malloc(sizeof(Edge));
258 if (!edge) {
259 // Cleanup on allocation failure
260 for (int k = 0; k < index; k++) {
261 free(edges[k]);
262 }
263 free(edges);
264 free(neighbors);
265 free(nodes);
266 return NULL;
267 }
268
269 edge->source = nodes[i];
270 edge->dest = neighbors[j];
271 edge->label = graph_get_label(gr, nodes[i], neighbors[j]);
272 edges[index++] = edge;
273 }
274 free(neighbors);
275 }
276 }
277 free(nodes);
278
279 edges[index] = NULL; // NULL terminate array
280 return edges;
281}

327void graph_free(Graph* gr) {
328 if (!gr) return;
329
330 // Free all nodes and their adjacency lists
331 if (gr->nodes) {
332 for (int i = 0; i < gr->nodes->bucket_count; i++) {
333 HashNode* entry = gr->nodes->buckets[i];
334 while (entry) {
335 // Free node key (strdup'd in add_node)
336 free((void*)entry->key);
337
338 // Free adjacency list and its contents
339 if (entry->value) {
340 HashTable* adj = (HashTable*)entry->value;
341 if (gr->is_labelled) {
342 // Free edge labels
343 void** neighbors = hash_table_keyset(adj);
344 if (neighbors) {
345 for (int j = 0; neighbors[j] != NULL; j++) {
346 void* label = hash_table_get(adj, neighbors[j]);
347 free(label);
348 }
349 free(neighbors);
350 }
351 }
352 hash_table_free(adj);
353 }
354 entry = entry->next;
355 }
356 }
357 hash_table_free(gr->nodes);
358 }
359
360 free(gr);
