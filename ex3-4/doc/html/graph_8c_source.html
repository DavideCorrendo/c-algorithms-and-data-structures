60Bool graph_add_node(Graph* gr, const void* node) {
61 if (!gr || !node) return false;
62
63 // Check if node already exists
64 if (graph_contains_node(gr, node)) return false;
65

69

76

78 hash_table_put(gr->nodes, node_copy, adjacency_list);
79 return true;

91Bool graph_add_edge(Graph* gr, const void* node1, const void* node2, const void* label) {
92 if (!gr || !node1 || !node2) return false;
93 if (gr->is_labelled && !label) return false;
94
95 // Get adjacency lists for both nodes
96 HashTable* adj1 = hash_table_get(gr->nodes, node1);
97 HashTable* adj2 = hash_table_get(gr->nodes, node2);
98 if (!adj1 || !adj2) return false;
99
100 // Check if edge already exists
101 if (hash_table_contains_key(adj1, node2)) {
102 return true; // Edge already exists
103 }
104
105 // Create copy of label if graph is labelled
106 void* label_copy = NULL;
107 if (gr->is_labelled) {
108 label_copy = malloc(sizeof(float));
109 if (!label_copy) return false;
110 memcpy(label_copy, label, sizeof(float));
111 }
112
113 // Add forward edge
114 hash_table_put(adj1, node2, label_copy);
115
116 // For undirected graphs, add reverse edge
117 if (!gr->is_directed) {
118 void* reverse_label = NULL;
119 if (gr->is_labelled) {
120 reverse_label = malloc(sizeof(float));
121 if (!reverse_label) {
122 // Cleanup if allocation fails
123 hash_table_remove(adj1, node2);
124 free(label_copy);
125 return false;
126 }
127 memcpy(reverse_label, label, sizeof(float));
128 }
129 hash_table_put(adj2, node1, reverse_label);
130 }
131
132 gr->edge_count++;
133 return true;
134}

167Bool graph_remove_node(Graph* gr, const void* node) {
168 if (!gr || !node) return false;
169
170 HashTable* adj = hash_table_get(gr->nodes, node);
171 if (!adj) return false;
172
173 // Remove all edges connected to this node
174 void** neighbors = hash_table_keyset(adj);
175 if (neighbors) {
176 for (int i = 0; neighbors[i] != NULL; i++) {
177 graph_remove_edge(gr, node, neighbors[i]);
178 if (!gr->is_directed) {
179 graph_remove_edge(gr, neighbors[i], node);
180 }
181 }
182 free(neighbors);
183 }
184
185 // Free adjacency list and remove node
186 hash_table_free(adj);
187 hash_table_remove(gr->nodes, node);
188
189 return true;
190}

199Bool graph_remove_edge(Graph* gr, const void* node1, const void* node2) {

265Edge** graph_get_edges(const Graph* gr) {
266 if (!gr || gr->edge_count == 0) return NULL;
267
268 // Allocate array for edges
269 Edge** edges = malloc((gr->edge_count + 1) * sizeof(Edge*));
270 if (!edges) return NULL;
271
272 int index = 0;
273 void** nodes = hash_table_keyset(gr->nodes);
274 if (!nodes) {
275 free(edges);
276 return NULL;
277 }
278
279 // Iterate through all nodes and their edges
280 for (int i = 0; nodes[i] != NULL; i++) {
281 HashTable* adj = hash_table_get(gr->nodes, nodes[i]);
282 void** neighbors = hash_table_keyset(adj);
283
284 if (neighbors) {
285 for (int j = 0; neighbors[j] != NULL; j++) {
286 Edge* edge = malloc(sizeof(Edge));
287 if (!edge) {
288 // Cleanup on allocation failure
289 for (int k = 0; k < index; k++) {
290 free(edges[k]);
291 }
292 free(edges);
293 free(neighbors);
294 free(nodes);
295 return NULL;
296 }
297
298 edge->source = nodes[i];
299 edge->dest = neighbors[j];
300 edge->label = graph_get_label(gr, nodes[i], neighbors[j]);
301 edges[index++] = edge;
302 }
303 free(neighbors);
304 }
305 }
306 free(nodes);
307
308 edges[index] = NULL; // NULL terminate array
309 return edges;
310}

356void graph_free(Graph* gr) {
357 if (!gr) return;
358
359 // Free all nodes and their adjacency lists
360 if (gr->nodes) {
361 for (int i = 0; i < gr->nodes->bucket_count; i++) {
362 HashNode* entry = gr->nodes->buckets[i];
363 while (entry) {
364 // Free node key (strdup'd in add_node)
365 free((void*)entry->key);
366
367 // Free adjacency list and its contents
368 if (entry->value) {
369 HashTable* adj = (HashTable*)entry->value;
370 if (gr->is_labelled) {
371 // Free edge labels
372 void** neighbors = hash_table_keyset(adj);
373 if (neighbors) {
374 for (int j = 0; neighbors[j] != NULL; j++) {
375 void* label = hash_table_get(adj, neighbors[j]);
376 free(label);
377 }
378 free(neighbors);
379 }
380 }
381 hash_table_free(adj);
382 }
383 entry = entry->next;
384 }
385 }
386 hash_table_free(gr->nodes);
387 }
388
389 free(gr);
390}
