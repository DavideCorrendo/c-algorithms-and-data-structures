180void quick_sort_rec(void **arr, int left, int right, int (*compar)(const void *, const void *)) {
181 if (left >= right) return;
182
183 void *pivot = arr[(left + right) / 2];
184 int i = left, j = right;
185
186 while (i <= j) {
187 while (compar(arr[i], pivot) < 0) i++;
188 while (compar(arr[j], pivot) > 0) j--;
189 if (i <= j) {
190 swap(&arr[i], &arr[j]);
191 i++;
192 j--;
193 }
194 }
195
196 if (left < j) quick_sort_rec(arr, left, j, compar);
197 if (i < right) quick_sort_rec(arr, i, right, compar);
198}
